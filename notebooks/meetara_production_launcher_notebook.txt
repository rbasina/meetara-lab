        batch_size = 2
    elif "V100" in gpu_name:

import subprocess
import shutil
from pathlib import Path

llama_path = Path("llama.cpp")
if not (llama_path / "CMakeLists.txt").exists():
    print("📥 Cloning llama.cpp...")
    if llama_path.exists():
        shutil.rmtree(llama_path)
    subprocess.run(["git", "clone", "https://github.com/ggerganov/llama.cpp.git"], check=True)

# Build llama.cpp with CUDA
os.chdir("llama.cpp")

# Clean build directory
build_dir = Path("build")
if build_dir.exists():
    shutil.rmtree(build_dir)

# Configure with CUDA
print("🔧 Configuring CMake with CUDA...")
cmake_cmd = ["cmake", "-B", "build", "-DCMAKE_BUILD_TYPE=Release", "-DGGML_CUDA=ON"]
subprocess.run(cmake_cmd, check=True)

# Build
print("🔨 Building llama.cpp...")
build_cmd = ["cmake", "--build", "build", "--config", "Release", "-j"]
subprocess.run(build_cmd, check=True)

# 4. Apply memory and config fixes
print("🔧 Applying memory and config fixes...")

# Create a simple fix script
fix_script = '''
import os
import sys
from pathlib import Path

# Add project root to path
project_root = Path.cwd()
sys.path.insert(0, str(project_root))

# Fix max_memory parameter in model factory
model_factory_path = Path("trinity_core/agents/model_factory.py")
if model_factory_path.exists():
    with open(model_factory_path, 'r') as f:
        content = f.read()
    
    # Fix max_memory parameter
    content = content.replace('max_memory=model_loading_config.get(\'max_memory\', "auto")', 'max_memory=None  # Fix: Use None instead of "auto"')
    content = content.replace('max_memory="auto"', 'max_memory=None  # Fix: Use None instead of "auto"')
    
    with open(model_factory_path, 'w') as f:
        f.write(content)
    
    print("✅ Fixed max_memory parameter in model factory")

# Add get_config fallback to config manager
config_manager_path = Path("trinity_core/core_components/config_manager.py")
if config_manager_path.exists():
    with open(config_manager_path, 'r') as f:
        content = f.read()
    
    if 'def get_config(' not in content:
        # Add get_config method after get_config_dict
        lines = content.split('\\n')
        for i, line in enumerate(lines):
            if line.strip() == 'def get_config_dict(self):':
                # Find end of method
                for j in range(i + 1, len(lines)):
                    if lines[j].strip().startswith('def ') and lines[j].strip() != 'def get_config_dict(self):':
                        break
                    elif lines[j].strip() == '' and j > i + 5:
                        break
                
                # Insert get_config method
                get_config_method = '''
    def get_config(self):
        """
        Fallback method for backward compatibility.
        Returns the same as get_config_dict().
        """
        return self.get_config_dict()
'''
                lines.insert(j, get_config_method)
                content = '\\n'.join(lines)
                break
        
        with open(config_manager_path, 'w') as f:
            f.write(content)
        
        print("✅ Added get_config fallback method")

print("✅ All fixes applied successfully")
'''

# Write and execute the fix script
with open("apply_fixes.py", "w") as f:
    f.write(fix_script)
